		id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
		name VARCHAR(190) NOT NULL,
		trigger_key VARCHAR(40) NOT NULL,

		lead_id BIGINT UNSIGNED NOT NULL DEFAULT 0,

		status VARCHAR(20) NOT NULL,
		trigger_key VARCHAR(40) DEFAULT '' NOT NULL,
		template_name VARCHAR(190) DEFAULT '' NOT NULL,

		PRIMARY KEY (id),
		KEY lead_id (lead_id),

function gida_admin_suite_maybe_upgrade_whatsapp_templates_table() {
	global $wpdb;
	$table = $wpdb->prefix . 'gida_whatsapp_templates';
	$exists = $wpdb->get_var( $wpdb->prepare( 'SHOW TABLES LIKE %s', $table ) ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	if ( $exists !== $table ) {
		return;
	}
	$columns = $wpdb->get_col( "SHOW COLUMNS FROM {$table}", 0 ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	if ( ! is_array( $columns ) ) {
		return;
	}
	if ( ! in_array( 'html_source', $columns, true ) ) {
		$wpdb->query( "ALTER TABLE {$table} ADD COLUMN html_source LONGTEXT NULL AFTER template_body" ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	}
	if ( ! in_array( 'media_url', $columns, true ) ) {
		$wpdb->query( "ALTER TABLE {$table} ADD COLUMN media_url TEXT NULL AFTER template_body" ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	}
	if ( ! in_array( 'sequence_no', $columns, true ) ) {
		$wpdb->query( "ALTER TABLE {$table} ADD COLUMN sequence_no INT NOT NULL DEFAULT 1 AFTER html_source" ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	}
	if ( ! in_array( 'content_sid', $columns, true ) ) {
		$wpdb->query( "ALTER TABLE {$table} ADD COLUMN content_sid VARCHAR(80) NOT NULL DEFAULT ''" ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	}
}
add_action( 'init', 'gida_admin_suite_maybe_upgrade_whatsapp_templates_table', 20 );

function gida_admin_suite_get_current_panel() {
	$panel = get_query_var( 'gida_panel' );
	if ( $panel ) {
		return (string) $panel;
	}
	$request_path = trim( (string) wp_parse_url( $_SERVER['REQUEST_URI'] ?? '', PHP_URL_PATH ), '/' );
	if ( preg_match( '#(^|/)admin-login$#', $request_path ) ) {
		return 'login';
	}
	if ( preg_match( '#(^|/)admin-register$#', $request_path ) ) {
		return 'register';
	}
	if ( preg_match( '#(^|/)dashboard$#', $request_path ) ) {
		return 'dashboard';
	}
	if ( preg_match( '#(^|/)my-dashboard$#', $request_path ) ) {
		return 'customer_dashboard';
	}
	return '';
}

function gida_admin_suite_handle_routes() {
	$panel = gida_admin_suite_get_current_panel();
	if ( ! in_array( $panel, array( 'login', 'register', 'dashboard', 'customer_dashboard' ), true ) ) {
		return;
	}

	if ( 'POST' === $_SERVER['REQUEST_METHOD'] ) {
		gida_brnd_cms_maybe_process_save();
		if ( isset( $_POST['gida_admin_login_action'] ) ) {
			gida_admin_suite_process_login();
		}
		if ( isset( $_POST['gida_admin_register_send_otp'] ) ) {
			gida_admin_suite_process_register_send_otp();
		}
		if ( isset( $_POST['gida_admin_register_verify_otp'] ) ) {
			gida_admin_suite_process_register_verify_otp();
		}
	}

	if ( 'login' === $panel ) {
		gida_admin_suite_render_login();
		exit;
	}
	if ( 'register' === $panel ) {
		gida_admin_suite_render_register();
		exit;
	}

	if ( ! is_user_logged_in() ) {
		wp_safe_redirect( home_url( '/admin-login/' ) );
		exit;
	}

	if ( current_user_can( 'manage_options' ) ) {
		if ( 'customer_dashboard' === $panel ) {
			wp_safe_redirect( home_url( '/dashboard/' ) );
			exit;
		}
		gida_admin_suite_render_dashboard();
		exit;
	}

	if ( 'dashboard' === $panel ) {
		wp_safe_redirect( home_url( '/my-dashboard/' ) );
		exit;
	}
	gida_admin_suite_render_customer_dashboard();
	exit;
}
add_action( 'template_redirect', 'gida_admin_suite_handle_routes' );

function gida_admin_suite_process_login() {
	if ( empty( $_POST['gida_admin_login_nonce'] ) || ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['gida_admin_login_nonce'] ) ), 'gida_admin_login' ) ) {
		return;
	}

	$username = isset( $_POST['log'] ) ? sanitize_text_field( wp_unslash( $_POST['log'] ) ) : '';
	$password = isset( $_POST['pwd'] ) ? (string) wp_unslash( $_POST['pwd'] ) : '';
	if ( is_email( $username ) ) {
		$user_by_email = get_user_by( 'email', $username );
		if ( $user_by_email && ! empty( $user_by_email->user_login ) ) {
			$username = (string) $user_by_email->user_login;
		}
	}

	$result = wp_signon(
		array(
			'user_login'    => $username,
			'user_password' => $password,
			'remember'      => true,
		),
		is_ssl()
	);

	if ( is_wp_error( $result ) ) {
		wp_safe_redirect( add_query_arg( 'login', 'failed', home_url( '/admin-login/' ) ) );
		exit;
	}

	wp_safe_redirect( gida_admin_suite_get_post_login_redirect( $result->ID ) );
	exit;
}

function gida_admin_suite_get_safe_redirect_from_request() {
	$raw_redirect = isset( $_REQUEST['redirect_to'] ) ? (string) wp_unslash( $_REQUEST['redirect_to'] ) : '';
	if ( '' === $raw_redirect ) {
		return '';
	}
	return wp_validate_redirect( $raw_redirect, '' );
}

function gida_admin_suite_get_user_purchase_url( $user_id ) {
	$page_id = absint( get_user_meta( $user_id, '_gida_subscription_page_id', true ) );
	if ( $page_id > 0 ) {
		$link = get_permalink( $page_id );
		if ( $link ) {
			return $link;
		}
	}

	$page_url = (string) get_user_meta( $user_id, '_gida_subscription_page_url', true );
	if ( $page_url ) {
		$validated = wp_validate_redirect( $page_url, '' );
		if ( $validated ) {
			return $validated;
		}
	}

	$user = get_userdata( $user_id );
	if ( ! $user || empty( $user->user_email ) ) {
		return '';
	}

	$q = new WP_Query(
		array(
			'post_type'      => 'gida_booking',
			'post_status'    => 'publish',
			'posts_per_page' => 1,
			'orderby'        => 'date',
			'order'          => 'DESC',
			'meta_query'     => array(
				array(
					'key'     => '_gida_email',
					'value'   => (string) $user->user_email,
					'compare' => '=',
				),
			),
		)
	);
	if ( ! $q->have_posts() ) {
		return '';
	}
	$q->the_post();
	$booking_id = get_the_ID();
	wp_reset_postdata();

	$service_name = (string) get_post_meta( $booking_id, '_gida_service', true );
	if ( ! $service_name ) {
		return '';
	}

	$mapped_url = gida_admin_suite_map_service_name_to_url( $service_name );
	if ( $mapped_url ) {
		update_user_meta( $user_id, '_gida_subscription_page_url', $mapped_url );
	}
	return $mapped_url;
}

function gida_admin_suite_map_service_name_to_url( $service_name ) {
	$service_name = trim( (string) $service_name );
	if ( '' === $service_name ) {
		return '';
	}

	$map = array(
		'Meta Ads Management'   => home_url( '/service-1/' ),
		'Google Ads Management' => home_url( '/service-2/' ),
		'WhatsApp Ads API'      => home_url( '/service-3/' ),
		'WhatsApp Automation'   => home_url( '/service-4/' ),
	);
	$dynamic = get_posts(
		array(
			'post_type'      => 'gida_service',
			'post_status'    => 'publish',
			'posts_per_page' => 200,
		)
	);
	foreach ( $dynamic as $service_post ) {
		$map[ $service_post->post_title ] = gida_admin_suite_get_service_frontend_url( (int) $service_post->ID );
	}

	return isset( $map[ $service_name ] ) ? (string) $map[ $service_name ] : '';
}

function gida_admin_suite_get_service_frontend_url( $service_id ) {
	$service_id = absint( $service_id );
	if ( $service_id <= 0 || 'gida_service' !== get_post_type( $service_id ) ) {
		return '';
	}
	$template_slug = 'gida-service-page-' . $service_id . '.php';
	$mapped_pages  = get_posts(
		array(
			'post_type'      => 'page',
			'post_status'    => 'publish',
			'posts_per_page' => 1,
			'meta_key'       => '_wp_page_template',
			'meta_value'     => $template_slug,
			'orderby'        => 'date',
			'order'          => 'DESC',
		)
	);
	if ( ! empty( $mapped_pages ) ) {
		$page_link = get_permalink( (int) $mapped_pages[0]->ID );
		if ( $page_link ) {
			return (string) $page_link;
		}
	}
	$service_link = get_permalink( $service_id );
	return $service_link ? (string) $service_link : '';
}

function gida_admin_suite_get_pending_registration_key( $email ) {
	return 'gida_admin_pending_registration_' . md5( strtolower( trim( (string) $email ) ) );
}

function gida_admin_suite_get_pending_registration_data( $email ) {
	if ( ! $email || ! is_email( $email ) ) {
		return array();
	}
	$data = get_transient( gida_admin_suite_get_pending_registration_key( $email ) );
	return is_array( $data ) ? $data : array();
}

function gida_admin_suite_process_register_send_otp() {
	if ( empty( $_POST['gida_admin_register_nonce'] ) || ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['gida_admin_register_nonce'] ) ), 'gida_admin_register_otp' ) ) {
		return;
	}

	$name     = isset( $_POST['register_name'] ) ? sanitize_text_field( wp_unslash( $_POST['register_name'] ) ) : '';
	$email    = isset( $_POST['register_email'] ) ? sanitize_email( wp_unslash( $_POST['register_email'] ) ) : '';
	$phone    = isset( $_POST['register_phone'] ) ? sanitize_text_field( wp_unslash( $_POST['register_phone'] ) ) : '';
	$password = isset( $_POST['register_password'] ) ? (string) wp_unslash( $_POST['register_password'] ) : '';

	if ( ! is_email( $email ) || '' === $name || '' === $phone || strlen( $password ) < 6 ) {
		wp_safe_redirect( add_query_arg( 'register', 'otp_expired', home_url( '/admin-register/' ) ) );
		exit;
	}
	if ( email_exists( $email ) ) {
		wp_safe_redirect( add_query_arg( 'register', 'exists', home_url( '/admin-register/' ) ) );
		exit;
	}

	$otp = (string) wp_rand( 100000, 999999 );
	set_transient(
		gida_admin_suite_get_pending_registration_key( $email ),
		array(
			'name'        => $name,
			'email'       => $email,
			'phone'       => $phone,
			'password'    => $password,
			'otp'         => $otp,
			'redirect_to' => gida_admin_suite_get_safe_redirect_from_request(),
		),
		15 * MINUTE_IN_SECONDS
	);

	wp_mail(
		$email,
		'Your OTP for account creation',
		"Hi {$name},\n\nYour OTP is: {$otp}\nThis OTP is valid for 15 minutes.\n\nThanks."
	);

	wp_safe_redirect(
		add_query_arg(
			array(
				'register' => 'otp_sent',
				'email'    => rawurlencode( $email ),
			),
			home_url( '/admin-register/' )
		)
	);
	exit;
}

function gida_admin_suite_process_register_verify_otp() {
	if ( empty( $_POST['gida_admin_register_nonce'] ) || ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['gida_admin_register_nonce'] ) ), 'gida_admin_register_otp' ) ) {
		return;
	}

	$email = isset( $_POST['register_email'] ) ? sanitize_email( wp_unslash( $_POST['register_email'] ) ) : '';
	$otp   = isset( $_POST['register_otp'] ) ? sanitize_text_field( wp_unslash( $_POST['register_otp'] ) ) : '';
	$data  = get_transient( gida_admin_suite_get_pending_registration_key( $email ) );
	if ( ! is_array( $data ) ) {
		wp_safe_redirect( add_query_arg( 'register', 'otp_expired', home_url( '/admin-register/' ) ) );
		exit;
	}
	if ( empty( $data['otp'] ) || (string) $data['otp'] !== $otp ) {
		wp_safe_redirect(
			add_query_arg(
				array(
					'register' => 'otp_invalid',
					'email'    => rawurlencode( $email ),
				),
				home_url( '/admin-register/' )
			)
		);
		exit;
	}

	$username_base = sanitize_user( current( explode( '@', $email ) ), true );
	$username      = $username_base ? $username_base : 'user';
	$suffix        = 1;
	while ( username_exists( $username ) ) {
		$username = $username_base . $suffix;
		++$suffix;
	}

	$user_id = wp_create_user( $username, (string) $data['password'], $email );
	if ( is_wp_error( $user_id ) ) {
		wp_safe_redirect( add_query_arg( 'register', 'exists', home_url( '/admin-register/' ) ) );
		exit;
	}

	wp_update_user(
		array(
			'ID'           => $user_id,
			'display_name' => (string) $data['name'],
			'first_name'   => (string) $data['name'],
		)
	);
	update_user_meta( $user_id, '_gida_phone', (string) $data['phone'] );
	$sent = gida_admin_suite_trigger_whatsapp( $user_id, (string) $data['name'], (string) $data['phone'], 'user_registered' );
	if ( $sent <= 0 ) {
		gida_admin_suite_trigger_whatsapp_content_template(
			$user_id,
			(string) $data['phone'],
			'user_registered',
			array(
				'name'     => (string) $data['name'],
				'services' => '',
				'amount'   => '',
			)
		);
	}
	delete_transient( gida_admin_suite_get_pending_registration_key( $email ) );

	wp_set_current_user( $user_id );
	wp_set_auth_cookie( $user_id );
	wp_safe_redirect( gida_admin_suite_get_post_login_redirect( $user_id ) );
	exit;
}

function gida_admin_suite_get_module() {
	$module = isset( $_GET['module'] ) ? sanitize_key( wp_unslash( $_GET['module'] ) ) : 'overview';
	$allowed = array(
		'overview',

		'users',
		'subscriptions',
		'services',
		'support',
		'leads',
		'whatsapp',
		'chat',
		'page_privacy',
		'page_terms',
		'page_refund',
		'page_contact',
	);
	if ( ! in_array( $module, $allowed, true ) ) {
		return 'overview';
	}
	return $module;
}

function gida_admin_suite_booking_status_label( $status ) {
	$status = strtolower( trim( (string) $status ) );
	if ( 'pending' === $status ) {
		$status = 'new';
	}
	if ( 'resolved' === $status ) {
		$status = 'contacted';
	}
	if ( ! in_array( $status, array( 'new', 'contacted' ), true ) ) {
		$status = 'new';
	}
	return 'new' === $status ? 'New' : 'Contacted';
}

function gida_admin_suite_render_customer_dashboard_content( $user_id, $section = 'bookings' ) {
	$user = get_userdata( $user_id );
	if ( ! $user ) {
		echo '<p>User not found.</p>';
		return;
	}

	$bookings = new WP_Query(
		array(
			'post_type'      => 'gida_booking',
			'posts_per_page' => 100,
			'post_status'    => 'publish',
			'orderby'        => 'date',
			'order'          => 'DESC',
			'meta_query'     => array(
				array(
					'key'     => '_gida_email',
					'value'   => (string) $user->user_email,
					'compare' => '=',
				),
			),
		)
	);

	$services = array();
	if ( $bookings->have_posts() ) {
		while ( $bookings->have_posts() ) {
			$bookings->the_post();
			$booking_id   = get_the_ID();
			$service_name = (string) get_post_meta( $booking_id, '_gida_service', true );
			$services[]   = $service_name;
		}
		wp_reset_postdata();
	}

	$services = array_values( array_unique( array_filter( $services ) ) );

	if ( 'bookings' === $section ) {
		echo '<h2>My Bookings</h2>';
		$bookings_list = new WP_Query(
			array(
				'post_type'      => 'gida_booking',
				'posts_per_page' => 100,
				'post_status'    => 'publish',
				'orderby'        => 'date',
				'order'          => 'DESC',
				'meta_query'     => array(
					array(
						'key'     => '_gida_email',
						'value'   => (string) $user->user_email,
						'compare' => '=',
					),
				),
			)
		);
		echo '<table class="gida-table"><thead><tr><th>Service</th><th>Date</th><th>Time</th><th>Status</th></tr></thead><tbody>';
		if ( $bookings_list->have_posts() ) {
			while ( $bookings_list->have_posts() ) {
				$bookings_list->the_post();
				$booking_id = get_the_ID();
				$status = gida_admin_suite_booking_status_label( get_post_meta( $booking_id, '_gida_booking_status', true ) );
				echo '<tr><td>' . esc_html( (string) get_post_meta( $booking_id, '_gida_service', true ) ) . '</td><td>' . esc_html( (string) get_post_meta( $booking_id, '_gida_meeting_date', true ) ) . '</td><td>' . esc_html( (string) get_post_meta( $booking_id, '_gida_meeting_time', true ) ) . '</td><td>' . esc_html( $status ) . '</td></tr>';
			}
			wp_reset_postdata();
		} else {
			echo '<tr><td colspan="4">No bookings found.</td></tr>';
		}
		echo '</tbody></table>';
		return;
	}

	if ( 'services' === $section ) {
		echo '<h2>My Services</h2>';
		$paid_records = gida_admin_suite_get_paid_subscription_records( $user->ID );
		if ( ! empty( $paid_records ) ) {
			echo '<div class="gida-paid-services-wrap">';
			foreach ( $paid_records as $record ) {
				echo '<article class="gida-paid-service-card">';
				echo '<div class="gida-paid-service-head"><strong>Subscribed Services</strong><span class="gida-paid-status">Success</span></div>';
				echo '<ul class="gida-paid-service-list">';
				foreach ( $record['selected_services'] as $service_row ) {
					$service_name = isset( $service_row['name'] ) ? (string) $service_row['name'] : '';
					if ( '' === trim( $service_name ) ) {
						continue;
					}
					$page_url = gida_admin_suite_map_service_name_to_url( $service_name );
					echo '<li><span>' . esc_html( $service_name ) . '</span>';
					if ( $page_url ) {
						echo '<a href="' . esc_url( $page_url ) . '" target="_blank" rel="noopener noreferrer">Open Service</a>';
					}
					echo '</li>';
				}
				echo '</ul>';
				echo '<div class="gida-paid-service-meta">';
				if ( ! empty( $record['total_amount'] ) ) {
					echo '<div><strong>Amount:</strong> INR ' . esc_html( number_format_i18n( (float) $record['total_amount'], 2 ) ) . '</div>';
				}
				if ( ! empty( $record['razorpay_payment_id'] ) ) {
					echo '<div><strong>Payment ID:</strong> ' . esc_html( (string) $record['razorpay_payment_id'] ) . '</div>';
				}
				if ( ! empty( $record['date'] ) ) {
					echo '<div><strong>Date:</strong> ' . esc_html( (string) $record['date'] ) . '</div>';
				}
				echo '</div>';
				echo '<div class="gida-paid-proof">';
				if ( ! empty( $record['payment_proof_url'] ) ) {

				} else {
					echo '<div class="gida-paid-proof-fallback">Payment proof not available.</div>';
				}
				echo '</div>';
				echo '</article>';
			}
			echo '</div>';
		} else {
			echo '<table class="gida-table"><thead><tr><th>Service</th><th>Page</th></tr></thead><tbody>';
			if ( ! empty( $services ) ) {
				foreach ( $services as $service_name ) {
					$page_url = gida_admin_suite_map_service_name_to_url( $service_name );
					$link     = $page_url ? '<a href="' . esc_url( $page_url ) . '" target="_blank" rel="noopener noreferrer">Open Service</a>' : 'N/A';
					echo '<tr><td>' . esc_html( $service_name ) . '</td><td>' . wp_kses_post( $link ) . '</td></tr>';
				}
			} else {
				echo '<tr><td colspan="2">No services found.</td></tr>';
			}
			echo '</tbody></table>';
		}
		return;
	}

	$phone = (string) get_user_meta( $user->ID, '_gida_phone', true );
	echo '<h2>Profile</h2>';
	echo '<table class="gida-table"><tbody>';

	echo '<tr><th>Total Services Used</th><td>' . esc_html( (string) count( $services ) ) . '</td></tr>';
	echo '</tbody></table>';
}

function gida_admin_suite_get_chat_conversations_ajax() {
	gida_admin_suite_verify_ajax();
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_send_json_error( array( 'message' => 'Unauthorized' ), 403 );
	}
	global $wpdb;
	$conv_table = $wpdb->prefix . 'gida_chat_conversations';
	$msg_table  = $wpdb->prefix . 'gida_chat_messages';
	$user_id    = get_current_user_id();
	$pins       = get_user_meta( $user_id, '_gida_live_chat_pins', true );
	$pins       = is_array( $pins ) ? array_values( array_filter( array_map( 'absint', $pins ) ) ) : array();
	$rows       = $wpdb->get_results(
		"SELECT c.id, c.visitor_name, c.visitor_email, c.last_message_at, c.created_at,
			(SELECT m.message FROM {$msg_table} m WHERE m.conversation_id = c.id ORDER BY m.id DESC LIMIT 1) AS last_message,
			(SELECT m.sender_label FROM {$msg_table} m WHERE m.conversation_id = c.id AND m.sender_type = 'user' ORDER BY m.id DESC LIMIT 1) AS latest_user_label,
			(SELECT m.id FROM {$msg_table} m WHERE m.conversation_id = c.id ORDER BY m.id DESC LIMIT 1) AS last_message_id
		FROM {$conv_table} c
		ORDER BY COALESCE(c.last_message_at, c.created_at) DESC
		LIMIT 200",
		ARRAY_A
	); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	$list       = array();
	$pinned     = array();
	$normal     = array();
	foreach ( $rows as $row ) {
		if ( '' === trim( (string) $row['visitor_name'] ) && ! empty( $row['visitor_email'] ) ) {
			$user = get_user_by( 'email', sanitize_email( (string) $row['visitor_email'] ) );
			if ( $user ) {
				$row['visitor_name'] = (string) $user->display_name;
			}
		}
		$conversation_id = (int) $row['id'];
		$last_message_id = (int) $row['last_message_id'];
		$last_read_id    = (int) get_user_meta( $user_id, '_gida_chat_read_' . $conversation_id, true );
		$row['unread']   = $last_message_id > $last_read_id ? 1 : 0;
		$row['is_pinned'] = in_array( $conversation_id, $pins, true ) ? 1 : 0;
		if ( $row['is_pinned'] ) {
			$pinned[] = $row;
		} else {
			$normal[] = $row;
		}
	}
	$list = array_merge( $pinned, $normal );
	wp_send_json_success( array( 'conversations' => $list ) );
}
add_action( 'wp_ajax_gida_chat_conversations', 'gida_admin_suite_get_chat_conversations_ajax' );

function gida_admin_suite_toggle_chat_pin_ajax() {
	gida_admin_suite_verify_ajax();
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_send_json_error( array( 'message' => 'Unauthorized' ), 403 );
	}
	$conversation_id = isset( $_POST['conversation_id'] ) ? absint( $_POST['conversation_id'] ) : 0;
	$pin             = isset( $_POST['pin'] ) ? absint( $_POST['pin'] ) : 0;
	if ( $conversation_id <= 0 ) {
		wp_send_json_error( array( 'message' => 'Invalid conversation' ), 400 );
	}
	$user_id = get_current_user_id();
	$pins    = get_user_meta( $user_id, '_gida_live_chat_pins', true );
	$pins    = is_array( $pins ) ? array_values( array_filter( array_map( 'absint', $pins ) ) ) : array();
	if ( 1 === $pin ) {
		if ( ! in_array( $conversation_id, $pins, true ) ) {
			if ( count( $pins ) >= 2 ) {
				wp_send_json_error( array( 'message' => 'Maximum 2 pinned chats allowed' ), 400 );
			}
			$pins[] = $conversation_id;
		}
	} else {
		$pins = array_values(
			array_filter(
				$pins,
				function( $id ) use ( $conversation_id ) {
					return (int) $id !== $conversation_id;
				}
			)
		);
	}
	update_user_meta( $user_id, '_gida_live_chat_pins', $pins );
	wp_send_json_success( array( 'pins' => $pins ) );
}

function gida_admin_suite_delete_chat_conversation_ajax() {
	gida_admin_suite_verify_ajax();
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_send_json_error( array( 'message' => 'Unauthorized' ), 403 );
	}
	$conversation_id = isset( $_POST['conversation_id'] ) ? absint( $_POST['conversation_id'] ) : 0;
	if ( $conversation_id <= 0 ) {
		wp_send_json_error( array( 'message' => 'Invalid conversation' ), 400 );
	}

	$ok = false;
	if ( function_exists( 'gida_live_chat_delete_conversation' ) ) {
		$ok = gida_live_chat_delete_conversation( $conversation_id );
	} else {
		global $wpdb;
		$msg_table  = $wpdb->prefix . 'gida_chat_messages';
		$conv_table = $wpdb->prefix . 'gida_chat_conversations';
		$wpdb->delete( $msg_table, array( 'conversation_id' => $conversation_id ), array( '%d' ) );
		$ok = (bool) $wpdb->delete( $conv_table, array( 'id' => $conversation_id ), array( '%d' ) );
	}
	if ( ! $ok ) {
		wp_send_json_error( array( 'message' => 'Delete failed' ), 500 );
	}

	$user_id = get_current_user_id();
	$pins    = get_user_meta( $user_id, '_gida_live_chat_pins', true );
	$pins    = is_array( $pins ) ? array_values( array_filter( array_map( 'absint', $pins ) ) ) : array();
	$pins    = array_values(
		array_filter(
			$pins,
			function( $id ) use ( $conversation_id ) {
				return (int) $id !== $conversation_id;
			}
		)
	);
	update_user_meta( $user_id, '_gida_live_chat_pins', $pins );

	delete_user_meta( $user_id, '_gida_chat_read_' . $conversation_id );
	wp_send_json_success();
}

function gida_admin_suite_reply_chat_ajax() {
	gida_admin_suite_verify_ajax();
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_send_json_error( array( 'message' => 'Unauthorized' ), 403 );
	}
	if ( ! function_exists( 'gida_live_chat_insert_message' ) ) {
		wp_send_json_error( array( 'message' => 'Live chat unavailable' ), 400 );
	}
	$conversation_id = isset( $_POST['conversation_id'] ) ? absint( $_POST['conversation_id'] ) : 0;
	$message         = isset( $_POST['message'] ) ? wp_kses_post( wp_unslash( $_POST['message'] ) ) : '';
	if ( $conversation_id <= 0 || '' === trim( wp_strip_all_tags( $message ) ) ) {
		wp_send_json_error( array( 'message' => 'Message is required' ), 400 );
	}
	$current_user = wp_get_current_user();
	$label        = $current_user && $current_user->display_name ? $current_user->display_name : 'Support';
	$item         = gida_live_chat_insert_message( $conversation_id, 'admin', $label, $message, 0 );
	if ( ! $item ) {
		wp_send_json_error( array( 'message' => 'Unable to send reply' ), 500 );
	}
	if ( function_exists( 'gida_live_chat_emit_event' ) ) {
		gida_live_chat_emit_event(
			'chat-message',
			array(
				'conversation_id' => $conversation_id,
				'message'         => $item,
			)
		);
	}
	update_user_meta( get_current_user_id(), '_gida_chat_read_' . $conversation_id, (int) $item['id'] );
	wp_send_json_success( array( 'message' => $item ) );
}

	update_post_meta( $plan_id, '_gida_plan_button_url', $button_url );

function gida_admin_suite_parse_selected_services( $raw_items ) {
	$decoded = json_decode( (string) $raw_items, true );
	$items   = array();
	if ( ! is_array( $decoded ) ) {
		return $items;
	}
	foreach ( $decoded as $item ) {
		if ( ! is_array( $item ) ) {
			continue;
		}
		$name  = isset( $item['name'] ) ? sanitize_text_field( (string) $item['name'] ) : '';
		$price = isset( $item['price'] ) ? (float) $item['price'] : 0;
		if ( '' === $name ) {
			continue;
		}
		$items[] = array(
			'name'  => $name,
			'price' => round( $price, 2 ),
		);
	}
	return $items;
}

function gida_admin_suite_append_subscription_history( $user_id, $entry ) {
	$user_id = absint( $user_id );
	if ( $user_id <= 0 || ! is_array( $entry ) ) {
		return;
	}
	$history = get_user_meta( $user_id, '_gida_subscription_history', true );
	if ( ! is_array( $history ) ) {
		$history = array();
	}
	$history[] = $entry;
	update_user_meta( $user_id, '_gida_subscription_history', array_slice( $history, -50 ) );
}

function gida_admin_suite_render_subscription_history_html( $history ) {
	if ( ! is_array( $history ) || empty( $history ) ) {
		return 'No history';
	}
	$history = array_reverse( $history );
	$output  = '';
	foreach ( $history as $entry ) {
		if ( ! is_array( $entry ) ) {
			continue;
		}
		$service_label = '';
		if ( ! empty( $entry['selected_services'] ) && is_array( $entry['selected_services'] ) ) {
			$names = array();
			foreach ( $entry['selected_services'] as $service_row ) {
				if ( is_array( $service_row ) && ! empty( $service_row['name'] ) ) {
					$names[] = sanitize_text_field( (string) $service_row['name'] );
				}
			}
			$service_label = implode( ', ', $names );
		}
		if ( '' === $service_label ) {
			$service_label = isset( $entry['service'] ) ? sanitize_text_field( (string) $entry['service'] ) : '';
		}
		$status = isset( $entry['payment_status'] ) ? sanitize_text_field( (string) $entry['payment_status'] ) : 'Success';
		$amount = isset( $entry['total_amount'] ) ? (float) $entry['total_amount'] : 0;
		$pid    = isset( $entry['razorpay_payment_id'] ) ? sanitize_text_field( (string) $entry['razorpay_payment_id'] ) : '';
		$date   = isset( $entry['date'] ) ? sanitize_text_field( (string) $entry['date'] ) : '';
		$output .= '<div style="margin-bottom:8px;padding:8px;border:1px solid #e4e7ec;border-radius:8px;background:#fff;">';
		$output .= '<div><strong>Services:</strong> ' . esc_html( '' !== $service_label ? $service_label : 'N/A' ) . '</div>';
		$output .= '<div><strong>Status:</strong> ' . esc_html( $status ) . '</div>';
		if ( $amount > 0 ) {
			$output .= '<div><strong>Amount:</strong> INR ' . esc_html( number_format_i18n( $amount, 2 ) ) . '</div>';
		}
		if ( '' !== $pid ) {
			$output .= '<div><strong>Payment ID:</strong> ' . esc_html( $pid ) . '</div>';
		}
		if ( '' !== $date ) {
			$output .= '<div><strong>Date:</strong> ' . esc_html( $date ) . '</div>';
		}
		$output .= '</div>';
	}
	return '' !== $output ? $output : 'No history';
}

function gida_admin_suite_get_paid_subscription_records( $user_id ) {
	$user_id = absint( $user_id );
	if ( $user_id <= 0 ) {
		return array();
	}
	$history = get_user_meta( $user_id, '_gida_subscription_history', true );
	if ( ! is_array( $history ) || empty( $history ) ) {
		return array();
	}
	$records = array();
	foreach ( $history as $entry ) {
		if ( ! is_array( $entry ) ) {
			continue;
		}
		$status = isset( $entry['payment_status'] ) ? strtolower( trim( (string) $entry['payment_status'] ) ) : '';
		if ( '' !== $status && 'success' !== $status ) {
			continue;
		}
		$selected_services = array();
		if ( ! empty( $entry['selected_services'] ) && is_array( $entry['selected_services'] ) ) {
			foreach ( $entry['selected_services'] as $service_row ) {
				if ( ! is_array( $service_row ) || empty( $service_row['name'] ) ) {
					continue;
				}
				$selected_services[] = array(
					'name'  => sanitize_text_field( (string) $service_row['name'] ),
					'price' => isset( $service_row['price'] ) ? (float) $service_row['price'] : 0,
				);
			}
		}
		if ( empty( $selected_services ) && ! empty( $entry['service'] ) ) {
			$selected_services[] = array(
				'name'  => sanitize_text_field( (string) $entry['service'] ),
				'price' => 0,
			);
		}
		if ( empty( $selected_services ) ) {
			continue;
		}
		$records[] = array(
			'selected_services'   => $selected_services,
			'total_amount'        => isset( $entry['total_amount'] ) ? (float) $entry['total_amount'] : 0,
			'payment_status'      => 'Success',
			'razorpay_payment_id' => isset( $entry['razorpay_payment_id'] ) ? sanitize_text_field( (string) $entry['razorpay_payment_id'] ) : '',
			'payment_proof_url'   => isset( $entry['payment_proof_url'] ) ? esc_url_raw( (string) $entry['payment_proof_url'] ) : '',
			'date'                => isset( $entry['date'] ) ? sanitize_text_field( (string) $entry['date'] ) : '',
		);
	}
	return array_reverse( $records );
}

function gida_admin_suite_create_razorpay_order_ajax() {
	if ( ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['nonce'] ?? '' ) ), 'gida_payment_nonce' ) ) {
		wp_send_json_error( array( 'message' => 'Invalid nonce.' ), 403 );
	}
	$plan_id        = isset( $_POST['plan_id'] ) ? absint( $_POST['plan_id'] ) : 0;
	$selected_items = gida_admin_suite_parse_selected_services( wp_unslash( $_POST['selected_items'] ?? '[]' ) );
	$total_amount   = isset( $_POST['total_amount'] ) ? (float) wp_unslash( $_POST['total_amount'] ) : 0;
	if ( $plan_id <= 0 || 'gida_plan' !== get_post_type( $plan_id ) ) {
		wp_send_json_error( array( 'message' => 'Invalid plan.' ), 400 );
	}
	if ( $total_amount <= 0 ) {
		wp_send_json_error( array( 'message' => 'Invalid payment amount.' ), 400 );
	}
	$total_paise = (int) round( $total_amount * 100 );
	$request     = wp_remote_post(
		'https://api.razorpay.com/v1/orders',
		array(
			'timeout' => 15,
			'headers' => array(
				'Authorization' => 'Basic ' . base64_encode( GIDA_RAZORPAY_KEY_ID . ':' . GIDA_RAZORPAY_KEY_SECRET ),
				'Content-Type'  => 'application/json',
			),
			'body'    => wp_json_encode(
				array(
					'amount'          => $total_paise,
					'currency'        => 'INR',
					'receipt'         => 'gida_plan_' . $plan_id . '_' . time(),
					'payment_capture' => 1,
					'notes'           => array(
						'plan_id'   => (string) $plan_id,
						'user_id'   => (string) get_current_user_id(),
						'item_count'=> (string) count( $selected_items ),
					),
				)
			),
		)
	);
	if ( is_wp_error( $request ) ) {
		wp_send_json_error( array( 'message' => $request->get_error_message() ), 500 );
	}
	$body = json_decode( (string) wp_remote_retrieve_body( $request ), true );
	if ( ! is_array( $body ) || empty( $body['id'] ) ) {
		wp_send_json_error( array( 'message' => 'Unable to create Razorpay order.' ), 500 );
	}
	$user = wp_get_current_user();
	$user_id = get_current_user_id();
	wp_send_json_success(
		array(
			'order_id'     => sanitize_text_field( (string) $body['id'] ),
			'key_id'       => GIDA_RAZORPAY_KEY_ID,
			'amount_paise' => $total_paise,
			'prefill'      => array(
				'name'    => $user_id > 0 ? (string) $user->display_name : '',
				'email'   => $user_id > 0 ? (string) $user->user_email : '',
				'contact' => $user_id > 0 ? (string) get_user_meta( $user_id, '_gida_phone', true ) : '',
			),
		)
	);
}

function gida_admin_suite_verify_razorpay_payment_ajax() {
	if ( ! wp_verify_nonce( sanitize_text_field( wp_unslash( $_POST['nonce'] ?? '' ) ), 'gida_payment_nonce' ) ) {
		wp_send_json_error( array( 'message' => 'Invalid nonce.' ), 403 );
	}
	$plan_id             = isset( $_POST['plan_id'] ) ? absint( $_POST['plan_id'] ) : 0;
	$total_amount        = isset( $_POST['total_amount'] ) ? (float) wp_unslash( $_POST['total_amount'] ) : 0;
	$selected_items      = gida_admin_suite_parse_selected_services( wp_unslash( $_POST['selected_items'] ?? '[]' ) );
	$razorpay_order_id   = isset( $_POST['razorpay_order_id'] ) ? sanitize_text_field( wp_unslash( $_POST['razorpay_order_id'] ) ) : '';
	$razorpay_payment_id = isset( $_POST['razorpay_payment_id'] ) ? sanitize_text_field( wp_unslash( $_POST['razorpay_payment_id'] ) ) : '';
	$razorpay_signature  = isset( $_POST['razorpay_signature'] ) ? sanitize_text_field( wp_unslash( $_POST['razorpay_signature'] ) ) : '';
	if ( $plan_id <= 0 || 'gida_plan' !== get_post_type( $plan_id ) ) {
		wp_send_json_error( array( 'message' => 'Invalid plan.' ), 400 );
	}
	if ( '' === $razorpay_order_id || '' === $razorpay_payment_id || '' === $razorpay_signature ) {
		wp_send_json_error( array( 'message' => 'Missing Razorpay payment details.' ), 400 );
	}
	$payload           = $razorpay_order_id . '|' . $razorpay_payment_id;
	$expected_signature = hash_hmac( 'sha256', $payload, GIDA_RAZORPAY_KEY_SECRET );
	if ( ! hash_equals( $expected_signature, $razorpay_signature ) ) {
		wp_send_json_error( array( 'message' => 'Payment signature verification failed.' ), 400 );
	}
	$user_id = get_current_user_id();
	if ( $user_id > 0 ) {
		$entry = array(
			'plan_id'              => $plan_id,
			'plan_name'            => get_the_title( $plan_id ),
			'selected_services'    => $selected_items,
			'total_amount'         => round( $total_amount, 2 ),
			'payment_status'       => 'Success',
			'razorpay_order_id'    => $razorpay_order_id,
			'razorpay_payment_id'  => $razorpay_payment_id,
			'date'                 => current_time( 'mysql' ),
		);
		gida_admin_suite_append_subscription_history( $user_id, $entry );
		update_user_meta( $user_id, '_gida_subscription_status', 'subscribed' );
		update_user_meta( $user_id, '_gida_is_subscribed', 1 );
		update_user_meta( $user_id, '_gida_last_payment_id', $razorpay_payment_id );
		update_user_meta( $user_id, '_gida_last_payment_amount', round( $total_amount, 2 ) );
		update_user_meta( $user_id, '_gida_last_payment_date', current_time( 'mysql' ) );
		$service_names = array();
		if ( is_array( $selected_items ) ) {
			foreach ( $selected_items as $line_item ) {
				if ( is_array( $line_item ) && ! empty( $line_item['name'] ) ) {
					$service_names[] = sanitize_text_field( (string) $line_item['name'] );
				}
			}
		}
		gida_admin_suite_trigger_whatsapp_content_template(
			$user_id,
			(string) get_user_meta( $user_id, '_gida_phone', true ),
			'payment_success',
			array(
				'name'     => (string) wp_get_current_user()->display_name,
				'services' => implode( ', ', $service_names ),
				'amount'   => 'INR ' . number_format( (float) $total_amount, 2, '.', '' ),
			)
		);
	}
	wp_send_json_success(
		array(
			'message'    => $user_id > 0 ? 'Payment successful.' : 'Payment successful. Subscription history is linked when user is logged in.',
			'payment_id' => $razorpay_payment_id,
		)
	);
}

add_action( 'wp_ajax_nopriv_gida_verify_razorpay_payment', 'gida_admin_suite_verify_razorpay_payment_ajax' );

function gida_admin_suite_save_template_ajax() {
	gida_admin_suite_verify_ajax();
	global $wpdb;
	$table       = $wpdb->prefix . 'gida_whatsapp_templates';
	$now         = current_time( 'mysql' );
	$template_id = isset( $_POST['template_id'] ) ? absint( $_POST['template_id'] ) : 0;
	$name        = isset( $_POST['template_name'] ) ? sanitize_text_field( wp_unslash( $_POST['template_name'] ) ) : '';
	$body        = isset( $_POST['template_body'] ) ? sanitize_textarea_field( wp_unslash( $_POST['template_body'] ) ) : '';
	$media_url   = isset( $_POST['media_url'] ) ? esc_url_raw( trim( (string) wp_unslash( $_POST['media_url'] ) ) ) : '';
	$html_source = isset( $_POST['html_source'] ) ? wp_kses_post( wp_unslash( $_POST['html_source'] ) ) : '';
	$sequence_no = isset( $_POST['sequence_no'] ) ? absint( $_POST['sequence_no'] ) : 1;
	$trigger     = isset( $_POST['trigger_key'] ) ? sanitize_key( wp_unslash( $_POST['trigger_key'] ) ) : 'user_registered';
	if ( ! in_array( $trigger, gida_admin_suite_whitelist_whatsapp_trigger_keys(), true ) ) {
		$trigger = 'user_registered';
	}
	$content_sid = isset( $_POST['content_sid'] ) ? preg_replace( '/\s+/', '', sanitize_text_field( wp_unslash( $_POST['content_sid'] ) ) ) : '';
	if ( '' !== $content_sid && ! preg_match( '/^HX[0-9a-f]{32}$/i', $content_sid ) ) {
		wp_send_json_error(
			array( 'message' => 'Invalid Twilio Content Template SID. Use HX followed by 32 hex characters (from Twilio Console), or leave empty for the site default for that event.' ),
			400
		);
	}
	if ( '' === $name || '' === $body ) {
		wp_send_json_error( array( 'message' => 'Template name and final message are required.' ), 400 );
	}
	if ( $sequence_no <= 0 ) {
		$sequence_no = 1;
	}
	if ( '' !== $media_url && '' === gida_admin_suite_prepare_twilio_media_url( $media_url ) ) {
		wp_send_json_error( array( 'message' => 'Invalid image URL for Twilio. Please use a direct public HTTPS image URL (not localhost/private/page URL).' ), 400 );
	}
	$data = array(
		'name'          => $name,
		'trigger_key'   => $trigger,
		'template_body' => $body,
		'media_url'     => $media_url,
		'html_source'   => $html_source,
		'sequence_no'   => $sequence_no,
		'active'        => 1,
		'content_sid'   => $content_sid,
		'updated_at'    => $now,
	);
	if ( $template_id > 0 ) {
		$wpdb->update(
			$table,
			$data,
			array( 'id' => $template_id ),
			array( '%s', '%s', '%s', '%s', '%s', '%d', '%d', '%s', '%s' ),
			array( '%d' )
		);
	} else {
		$data['created_at'] = $now;
		$wpdb->insert(
			$table,
			$data,
			array( '%s', '%s', '%s', '%s', '%s', '%d', '%d', '%s', '%s', '%s' )
		);
	}
	wp_send_json_success();
}
add_action( 'wp_ajax_gida_save_whatsapp_template', 'gida_admin_suite_save_template_ajax' );

/**
 * Admin: send approved content template to any WhatsApp number (manual test / outreach).
 *
 * If admin templates exist for user_registered, sends them in sequence via ContentSid.
 * Otherwise sends a single approved template with the given name and services.
 */
function gida_admin_suite_send_whatsapp_templates_custom_ajax() {
	gida_admin_suite_verify_ajax();
	if ( ! current_user_can( 'manage_options' ) ) {
		wp_send_json_error( array( 'message' => 'Unauthorized' ), 403 );
	}
	$phone    = isset( $_POST['phone'] ) ? sanitize_text_field( wp_unslash( $_POST['phone'] ) ) : '';
	$name     = isset( $_POST['recipient_name'] ) ? sanitize_text_field( wp_unslash( $_POST['recipient_name'] ) ) : '';
	$services = isset( $_POST['recipient_services'] ) ? sanitize_text_field( wp_unslash( $_POST['recipient_services'] ) ) : '';
	if ( '' === trim( $phone ) ) {
		wp_send_json_error( array( 'message' => 'WhatsApp number is required.' ), 400 );
	}
	if ( '' === gida_admin_suite_normalize_whatsapp_to( $phone ) ) {
		wp_send_json_error( array( 'message' => 'Invalid WhatsApp number. Use country code or 10-digit Indian number.' ), 400 );
	}

	$sent = gida_admin_suite_trigger_whatsapp( 0, $name, $phone, 'user_registered', $services );

	if ( $sent <= 0 ) {
		$template_sid = gida_admin_suite_get_whatsapp_content_template_sid( 'user_registered' );
		if ( '' === $template_sid ) {
			wp_send_json_error( array( 'message' => 'No approved template SID configured. Cannot send.' ), 400 );
		}
		$variables = gida_admin_suite_build_twilio_content_variables(
			'user_registered',
			array(
				'name'     => $name,
				'services' => $services,
				'amount'   => '',
			)
		);
		$result = gida_admin_suite_send_twilio_whatsapp_content_template( 0, $phone, 'user_registered', $template_sid, $variables );
		$sent   = $result > 0 ? 1 : 0;
	}

	if ( $sent <= 0 ) {
		wp_send_json_error( array( 'message' => 'Failed to send WhatsApp message. Check Twilio logs.' ), 400 );
	}
	wp_send_json_success( array( 'sent_count' => $sent ) );
}
add_action( 'wp_ajax_gida_send_whatsapp_templates_custom', 'gida_admin_suite_send_whatsapp_templates_custom_ajax' );

function gida_admin_suite_get_whatsapp_content_template_map() {
	$approved_sid = GIDA_TWILIO_WP_CONTENT_SID_DEFAULT;
	$defaults = array(
		'user_registered' => $approved_sid,
		'booking_created' => $approved_sid,
		'payment_success' => $approved_sid,
		'lead_created'    => $approved_sid,
		'follow_up'       => $approved_sid,
	);
	$stored = get_option( 'gida_twilio_template_sid_map', array() );
	if ( is_array( $stored ) ) {
		foreach ( $stored as $trigger_key => $sid ) {
			$sid = sanitize_text_field( (string) $sid );
			if ( '' !== $sid ) {
				$defaults[ sanitize_key( (string) $trigger_key ) ] = $sid;
			}
		}
	}
	return $defaults;
}

function gida_admin_suite_get_whatsapp_content_template_sid( $trigger_key ) {
	$trigger_key = sanitize_key( (string) $trigger_key );
	$map         = gida_admin_suite_get_whatsapp_content_template_map();
	return isset( $map[ $trigger_key ] ) ? sanitize_text_field( (string) $map[ $trigger_key ] ) : '';
}

function gida_admin_suite_build_twilio_content_variables( $trigger_key, $context = array() ) {
	$trigger_key = sanitize_key( (string) $trigger_key );
	$name        = isset( $context['name'] ) ? sanitize_text_field( (string) $context['name'] ) : '';
	$services    = isset( $context['services'] ) ? sanitize_text_field( (string) $context['services'] ) : '';
	$amount      = isset( $context['amount'] ) ? sanitize_text_field( (string) $context['amount'] ) : '';

	if ( '' === $name ) {
		$name = 'Customer';
	}

	/**
	 * Production-approved template for onboarding + booking notifications uses only WhatsApp placeholder {{1}} (customer name).
	 * Omit extra ContentVariables keys so Twilio accepts the payload (avoid 21656 / schema mismatch).
	 */
	if ( in_array( $trigger_key, array( 'user_registered', 'booking_created' ), true ) ) {
		return array(
			'1' => $name,
		);
	}

	$services_display = $services;
	if ( 'payment_success' === $trigger_key && '' !== $amount ) {
		$services_display = '' !== $services ? $services . ' (' . $amount . ')' : $amount;
	}
	if ( '' === $services_display ) {
		$services_display = 'Our Services';
	}

	return array(
		'1' => $name,
		'2' => $services_display,
	);
}

function gida_admin_suite_trigger_whatsapp_content_template( $lead_id, $phone, $trigger_key, $context = array() ) {
	$template_sid = gida_admin_suite_get_whatsapp_content_template_sid( $trigger_key );
	if ( '' === $template_sid ) {
		return 0;
	}
	$variables = gida_admin_suite_build_twilio_content_variables( $trigger_key, $context );
	return gida_admin_suite_send_twilio_whatsapp_content_template( $lead_id, $phone, $trigger_key, $template_sid, $variables );
}

/**
 * Send all active admin templates for a trigger to one phone via approved Twilio Content Template.
 *
 * Each row can set its own Twilio `content_sid` (HX…); otherwise the default for that trigger is used.
 * Order is `sequence_no` ascending.
 *
 * @param int    $lead_id     ID stored in WhatsApp logs (0 for manual / non-entity sends).
 * @param string $name        Value used for {{1}} (customer name) variable.
 * @param string $phone       Raw phone as entered or stored; normalized before send.
 * @param string $trigger_key Template group, e.g. user_registered, lead_created.
 * @param string $services    Optional services string for {{2}} variable.
 * @return int Number of messages Twilio accepted successfully (0 if none configured or all failed).
 */
function gida_admin_suite_trigger_whatsapp( $lead_id, $name, $phone, $trigger_key, $services = '' ) {
	global $wpdb;
	$template_table = $wpdb->prefix . 'gida_whatsapp_templates';
	$templates      = $wpdb->get_results( $wpdb->prepare( "SELECT * FROM {$template_table} WHERE trigger_key = %s AND active = 1 ORDER BY sequence_no ASC, id ASC", $trigger_key ), ARRAY_A ); // phpcs:ignore WordPress.DB.PreparedSQL.NotPrepared
	if ( empty( $templates ) ) {
		return 0;
	}

	$template_sid = gida_admin_suite_get_whatsapp_content_template_sid( $trigger_key );
	$sent_ok      = 0;

	foreach ( $templates as $template ) {
		$services_text = '' !== $services ? $services : sanitize_text_field( (string) $template['name'] );
		$variables     = gida_admin_suite_build_twilio_content_variables(
			$trigger_key,
			array(
				'name'     => $name,
				'services' => $services_text,
				'amount'   => '',
			)
		);

		$row_sid = isset( $template['content_sid'] ) ? trim( (string) $template['content_sid'] ) : '';
		$use_sid = ( '' !== $row_sid && preg_match( '/^HX[0-9a-f]{32}$/i', $row_sid ) ) ? $row_sid : $template_sid;

		if ( '' !== $use_sid ) {
			$sent_ok += gida_admin_suite_send_twilio_whatsapp_content_template(
				$lead_id,
				$phone,
				$trigger_key,
				$use_sid,
				$variables
			);
		} else {
			$message   = str_replace( '{{name}}', $name, (string) $template['template_body'] );
			$media_url = str_replace( '{{name}}', rawurlencode( $name ), (string) ( $template['media_url'] ?? '' ) );
			$sent_ok  += gida_admin_suite_send_twilio_whatsapp( $lead_id, $phone, $message, (string) $template['name'], $media_url );
		}
	}

	if ( 'lead_created' === $trigger_key ) {
		wp_schedule_single_event( time() + HOUR_IN_SECONDS * 12, 'gida_whatsapp_followup_event', array( $lead_id, $name, $phone ) );
	}

	return $sent_ok;
}

function gida_admin_suite_send_twilio_whatsapp( $lead_id, $phone, $message, $template_name, $media_url = '' ) {
	global $wpdb;
	$logs_table = $wpdb->prefix . 'gida_whatsapp_logs';
	$creds      = gida_admin_suite_twilio_credentials();
	$sid        = $creds['account_sid'];
	$token      = $creds['auth_token'];
	$from           = 'whatsapp:+918528779057';
	$phone          = gida_admin_suite_normalize_whatsapp_to( $phone );
	$original_media = trim( (string) $media_url );
	$media_url      = gida_admin_suite_prepare_twilio_media_url( $media_url );
	$message    = trim( (string) $message );

	$now    = current_time( 'mysql' );
	$status = 'failed';
	$resp   = '';

	if ( $sid && $token && $from && $phone && ( '' !== $message || '' !== $media_url ) ) {
		$body = array(
			'From' => $from,
			'To'   => $phone,
			'Body' => $message,
		);
		if ( '' !== $media_url ) {
			$body['MediaUrl'] = $media_url;
		}
		gida_admin_suite_twilio_messages_body_add_messaging_service( $body );
		$request = wp_remote_post(
			'https://api.twilio.com/2010-04-01/Accounts/' . rawurlencode( $sid ) . '/Messages.json',
			array(
				'headers' => array(
					'Authorization' => 'Basic ' . base64_encode( $sid . ':' . $token ),
				),
				'body'    => $body,
				'timeout' => 12,
			)
		);
		if ( ! is_wp_error( $request ) ) {
			$status = 200 === wp_remote_retrieve_response_code( $request ) || 201 === wp_remote_retrieve_response_code( $request ) ? 'sent' : 'failed';
			$resp   = wp_remote_retrieve_body( $request );
		} else {
			$resp = $request->get_error_message();
		}
	}

	if ( '' === $media_url && '' !== $original_media ) {
		$resp = trim( $resp . ' Media URL skipped: must be a public HTTPS URL accessible by Twilio (localhost/private URLs are not allowed).' );
	}

	$wpdb->insert(
		$logs_table,
		array(
			'lead_id'            => $lead_id,
			'phone'              => $phone,
			'direction'          => 'outgoing',
			'status'             => $status,
			'trigger_key'        => '',
			'template_name'      => $template_name,
			'template_sid'       => '',
			'content_variables'  => '',
			'message_text'       => '' !== $media_url ? ( '[image] ' . $media_url . "\n" . $message ) : $message,
			'provider_response'  => $resp,
			'created_at'         => $now,
		),
		array( '%d', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s' )
	);

	return 'sent' === $status ? 1 : 0;
}

function gida_admin_suite_infer_twilio_placeholder_keys_from_types_json( $data ) {
	if ( ! isset( $data['types'] ) || ! is_array( $data['types'] ) ) {
		return array();
	}

	$stringify = wp_json_encode( $data['types'] );
	if ( ! is_string( $stringify ) || '' === $stringify ) {
		return array();
	}

	preg_match_all( '/\{\{\s*([A-Za-z0-9_]{1,16})\s*\}\}/', $stringify, $matches );
	if ( empty( $matches[1] ) ) {
		return array();
	}

	$seen  = array();
	$order = array();
	foreach ( $matches[1] as $hit ) {
		$hit = (string) $hit;
		if ( ! preg_match( '/^[A-Za-z0-9_]{1,16}$/', $hit ) ) {
			continue;
		}
		$lk = strtolower( $hit );
		if ( isset( $seen[ $lk ] ) ) {
			continue;
		}
		$seen[ $lk ] = true;
		$order[]     = $hit;
	}
	return $order;
}

/**
 * Infer {{…}} placeholders from the entire Twilio Content JSON (types, channels, approvals, etc.).
 * Some WhatsApp approvals store body text outside `types`; this avoids wrong fallback keys → 21656.
 *
 * @param array|null $data Decoded Twilio GET /Content/{Sid} payload.
 * @return string[]
 */
function gida_admin_suite_infer_twilio_content_placeholders_from_full_json( $data ) {
	if ( ! is_array( $data ) ) {
		return array();
	}
	$hits   = array();
	$walker = null;
	$walker = function ( $node ) use ( &$walker, &$hits ) {
		if ( is_string( $node ) ) {
			if ( preg_match_all( '/\{\{\s*([A-Za-z0-9_]{1,16})\s*\}\}/', $node, $matches ) ) {
				foreach ( $matches[1] as $hit ) {
					$hits[] = (string) $hit;
				}
			}
			return;
		}
		if ( is_array( $node ) ) {
			foreach ( $node as $child ) {
				$walker( $child );
			}
		}
	};
	$walker( $data );

	$seen  = array();
	$order = array();
	foreach ( $hits as $hit ) {
		if ( ! preg_match( '/^[A-Za-z0-9_]{1,16}$/', $hit ) ) {
			continue;
		}
		$lk = strtolower( $hit );
		if ( isset( $seen[ $lk ] ) ) {
			continue;
		}
		$seen[ $lk ] = true;
		$order[]     = $hit;
	}
	return $order;
}

/**
 * Minimal JSON encoding for Messages API ContentVariables (UTF-8, no redundant escaping).
 *
 * @param array $assoc Assoc key => scalar.
 * @return string
 */
function gida_admin_suite_twilio_content_variables_json( $assoc ) {
	if ( ! is_array( $assoc ) ) {
		return '';
	}

	$n = json_encode( $assoc, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );
	if ( false !== $n ) {
		return $n;
	}

	$b = wp_json_encode( $assoc );
	return is_string( $b ) ? $b : '';
}

/**
 * GET Content definition and return declared/inferred placeholder keys (fixes Twilio REST 21656).
 *
 * @see https://www.twilio.com/docs/content/content-api-resources
 *
 * @param string $content_sid HX… template SID.
 * @return array{status:string,keys:string[]}
 */
function gida_admin_suite_fetch_twilio_content_variable_keys( $content_sid ) {
	$content_sid = trim( (string) $content_sid );
	$fallback    = array( 'status' => 'error', 'keys' => array() );

	if ( '' === $content_sid || ! preg_match( '/^HX[0-9a-f]{32}$/i', $content_sid ) ) {
		return $fallback;
	}

	// Bump cache when placeholder inference improves (avoid stale wrong keys stuck for 12h → 21656).
	$cache_key = 'gida_twilio_cvk_v4_' . md5( strtolower( $content_sid ) );
	$cached    = get_transient( $cache_key );
	if ( is_array( $cached ) && isset( $cached['status'], $cached['keys'] ) && is_array( $cached['keys'] ) ) {
		return $cached;
	}

	$creds = gida_admin_suite_twilio_credentials();
	$url   = 'https://content.twilio.com/v1/Content/' . rawurlencode( $content_sid );
	$resp  = wp_remote_get(
		$url,
		array(
			'timeout' => 15,
			'headers' => array(
				'Authorization' => 'Basic ' . base64_encode( $creds['account_sid'] . ':' . $creds['auth_token'] ),
				'Accept'        => 'application/json',
			),
		)
	);

	if ( is_wp_error( $resp ) || 200 !== (int) wp_remote_retrieve_response_code( $resp ) ) {
		set_transient( $cache_key, $fallback, MINUTE_IN_SECONDS * 30 );
		return $fallback;
	}

	$data = json_decode( (string) wp_remote_retrieve_body( $resp ), true );
	if ( ! is_array( $data ) ) {
		set_transient( $cache_key, $fallback, MINUTE_IN_SECONDS * 30 );
		return $fallback;
	}

	$vars = isset( $data['variables'] ) && is_array( $data['variables'] ) ? $data['variables'] : array();

	$good = array();
	foreach ( $vars as $key => $_ignored ) {
		$ks = (string) $key;
		if ( preg_match( '/^[A-Za-z0-9_]{1,16}$/', $ks ) ) {
			$good[] = $ks;
		}
	}

	if ( empty( $good ) ) {
		$good = gida_admin_suite_infer_twilio_placeholder_keys_from_types_json( $data );
	}
	if ( empty( $good ) ) {
		$good = gida_admin_suite_infer_twilio_content_placeholders_from_full_json( $data );
	}

	if ( empty( $good ) ) {
		$done = array( 'status' => 'empty_keys', 'keys' => array() );
		set_transient( $cache_key, $done, 12 * HOUR_IN_SECONDS );
		return $done;
	}

	$done = array(
		'status' => 'ok',
		'keys'   => $good,
	);
	set_transient( $cache_key, $done, 12 * HOUR_IN_SECONDS );
	return $done;
}

/**
 * Map name + services to ContentVariables matching template keys from Twilio.
 *
 * @param string[] $template_keys Sorted keys from Content API.
 * @param string   $name          Preferred slot {{1}}.
 * @param string   $services      Preferred slot {{2}}.
 * @return array<string,string>
 */
function gida_admin_suite_map_twilio_content_variables_to_keys( $template_keys, $name, $services ) {
	$name     = trim( (string) $name );
	$services = trim( (string) $services );

	if ( '' === $name ) {
		$name = 'Customer';
	}
	if ( '' === $services ) {
		$services = 'Our Services';
	}

	$combined = $name . ', ' . $services;

	if ( empty( $template_keys ) ) {
		return array(
			'1' => $name,
			'2' => $services,
		);
	}

	$n = count( $template_keys );
	if ( 1 === $n ) {
		return array(
			$template_keys[0] => $name,
		);
	}

	$slots = array( $name, $services );
	$out   = array();
	for ( $i = 0; $i < $n; $i++ ) {
		$out[ $template_keys[ $i ] ] = isset( $slots[ $i ] ) ? $slots[ $i ] : '-';
	}
	return $out;
}

/**
 * Normalize values for Twilio ContentVariables / WhatsApp (avoids REST error 21656).
 *
 * @link https://www.twilio.com/docs/api/errors/21656
 * @link https://www.twilio.com/docs/content/using-variables-with-content-api
 *
 * WhatsApp-approved templates disallow newlines inside variable values and reject
 * more than four consecutive spaces. Empty substitution values often fail validation.
 *
 * Important: Keys (e.g. "1","2") must match every {{n}} placeholder on the Content SID;
 * omit extra keys if the template uses only {{1}}.
 *
 * @param array $variables Assoc map of placeholder key => string value.
 * @return array Normalized assoc map.
 */
function gida_admin_suite_normalize_twilio_content_variables_for_send( $variables ) {
	if ( ! is_array( $variables ) ) {
		return array();
	}
	$out = array();
	foreach ( $variables as $key => $value ) {
		$key = (string) $key;
		if ( ! preg_match( '/^[A-Za-z0-9_]{1,16}$/', $key ) ) {
			continue;
		}
		$value          = (string) $value;
		$value          = preg_replace( '/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]/', '', $value );
		$value          = preg_replace( '/\R/u', ' ', $value ); // newline / line-break unicode
		$value          = str_replace( "\t", ' ', $value );
		$value          = preg_replace( '/ {5,}/u', ' ', $value ); // no 5+ consecutive spaces (Twilio)
		$value          = preg_replace( '/ +/u', ' ', $value );
		$value          = trim( $value );
		if ( '' === $value ) {
			$value = '-';
		}
		if ( function_exists( 'mb_strlen' ) && function_exists( 'mb_substr' ) && mb_strlen( $value, 'UTF-8' ) > 1600 ) {
			$value = mb_substr( $value, 0, 1600, 'UTF-8' );
		} elseif ( strlen( $value ) > 1600 ) {
			$value = substr( $value, 0, 1600 );
		}
		$out[ $key ] = $value;
	}
	return $out;
}

/**
 * When Content API key discovery fails or returns no keys, build a safe ContentVariables map.
 * Registration and booking use the approved single-placeholder template ({{1}} = customer name only).
 *
 * @param string $trigger_key  Trigger key.
 * @param string $name         Customer name.
 * @param string $services     Services / secondary slot text.
 * @return array<string,string>
 */
function gida_admin_suite_twilio_content_variables_fallback_projection( $trigger_key, $name, $services ) {
	$trigger_key = sanitize_key( (string) $trigger_key );
	$name        = trim( (string) $name );
	if ( '' === $name ) {
		$name = 'Customer';
	}
	if ( in_array( $trigger_key, array( 'user_registered', 'booking_created' ), true ) ) {
		return array( '1' => $name );
	}
	return gida_admin_suite_map_twilio_content_variables_to_keys( array(), $name, $services );
}

/**
 * Encode WP_Error for WhatsApp log provider_response (codes + error_data).
 *
 * @param WP_Error $error Error from wp_remote_*.
 * @return string JSON or empty string.
 */
function gida_admin_suite_format_wp_error_for_log( $error ) {
	if ( ! is_wp_error( $error ) ) {
		return '';
	}
	$payload = array(
		'codes' => $error->get_error_codes(),
	);
	$data_map = array();
	foreach ( $error->get_error_codes() as $code ) {
		$d = $error->get_error_data( $code );
		if ( null !== $d && '' !== $d && array() !== $d ) {
			$data_map[ $code ] = $d;
		}
	}
	if ( array() !== $data_map ) {
		$payload['error_data'] = $data_map;
	}
	$json = wp_json_encode( $payload, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES );
	return is_string( $json ) ? $json : '';
}

function gida_admin_suite_send_twilio_whatsapp_content_template( $lead_id, $phone, $trigger_key, $template_sid, $variables ) {
	global $wpdb;
	$logs_table = $wpdb->prefix . 'gida_whatsapp_logs';
	$creds      = gida_admin_suite_twilio_credentials();
	$sid        = $creds['account_sid'];
	$token      = $creds['auth_token'];
	$from       = 'whatsapp:+918528779057';
	$to         = gida_admin_suite_normalize_whatsapp_to( $phone );
	$now        = current_time( 'mysql' );
	$status     = 'failed';
	$resp       = '';

	if ( ! is_array( $variables ) ) {
		$variables = array();
	}
	$name_placeholder     = isset( $variables['1'] ) ? (string) $variables['1'] : 'Customer';
	$services_placeholder = isset( $variables['2'] ) ? (string) $variables['2'] : 'Our Services';
	if ( ! isset( $variables['1'] ) && array() !== $variables ) {
		foreach ( $variables as $vk => $vv ) {
			$vk = (string) $vk;
			if ( preg_match( '/^[A-Za-z0-9_]{1,16}$/', $vk ) && '' !== trim( (string) $vv ) ) {
				$name_placeholder = (string) $vv;
				break;
			}
		}
	}

	$content_sid_trim     = trim( (string) $template_sid );
	$content_var_cache_key = 'gida_twilio_cvk_v4_' . md5( strtolower( $content_sid_trim ) );

	$resolved = gida_admin_suite_fetch_twilio_content_variable_keys( $content_sid_trim );
	$proj     = array();
	if ( 'ok' === $resolved['status'] && ! empty( $resolved['keys'] ) ) {
		$proj = gida_admin_suite_map_twilio_content_variables_to_keys( $resolved['keys'], $name_placeholder, $services_placeholder );
	} else {
		$proj = gida_admin_suite_twilio_content_variables_fallback_projection( $trigger_key, $name_placeholder, $services_placeholder );
	}

	$norm_vars = gida_admin_suite_normalize_twilio_content_variables_for_send( $proj );
	$json_vars = empty( $norm_vars ) ? '' : gida_admin_suite_twilio_content_variables_json( $norm_vars );

	$response_notes = '';

	if ( $sid && $token && $from && $to && '' !== trim( (string) $template_sid ) ) {
		$twilio_endpoint = 'https://api.twilio.com/2010-04-01/Accounts/' . rawurlencode( $sid ) . '/Messages.json';
		$post_headers    = array(
			'Authorization' => 'Basic ' . base64_encode( $sid . ':' . $token ),
		);

		$body = array(
			'From'       => $from,
			'To'         => $to,
			'ContentSid' => trim( (string) $template_sid ),
		);
		if ( '' !== $json_vars ) {
			$body['ContentVariables'] = $json_vars;
		}
		gida_admin_suite_twilio_messages_body_add_messaging_service( $body );

		$request = wp_remote_post(
			$twilio_endpoint,
			array(
				'headers' => $post_headers,
				'body'    => $body,
				'timeout' => 12,
			)
		);
		if ( ! is_wp_error( $request ) ) {
			$code   = (int) wp_remote_retrieve_response_code( $request );
			$status = ( 200 === $code || 201 === $code ) ? 'sent' : 'failed';
			$resp   = (string) wp_remote_retrieve_body( $request );
			if ( 'sent' !== $status ) {
				$resp = 'HTTP ' . $code . ' | ' . $resp;
			}
		} else {
			$wp_err_json = gida_admin_suite_format_wp_error_for_log( $request );
			$resp        = 'WP_Error: ' . $request->get_error_message() . ( '' !== $wp_err_json ? ' | ' . $wp_err_json : '' );
		}

		if ( 'sent' !== $status && '' !== $json_vars ) {
			$t_err    = json_decode( (string) $resp, true );
			$is_21656 = strpos( $resp, '21656' ) !== false || ( is_array( $t_err ) && isset( $t_err['code'] ) && (int) $t_err['code'] === 21656 );
			if ( $is_21656 && count( $norm_vars ) > 1 ) {
				$solo_slot = '1';
				if ( 'ok' === $resolved['status'] && isset( $resolved['keys'][0] ) ) {
					$cand = (string) $resolved['keys'][0];
					if ( preg_match( '/^[A-Za-z0-9_]{1,16}$/', $cand ) ) {
						$solo_slot = $cand;
					}
				}

				if ( preg_match( '/^[A-Za-z0-9_]{1,16}$/', $solo_slot ) ) {
					$response_notes .= '[Attempt 1] ' . $resp . "\n\n";

					$tkey_san = sanitize_key( (string) $trigger_key );
					$merge_raw = in_array( $tkey_san, array( 'user_registered', 'booking_created' ), true )
						? trim( (string) $name_placeholder )
						: trim( (string) $name_placeholder . ', ' . (string) $services_placeholder );
					$retry_norm = gida_admin_suite_normalize_twilio_content_variables_for_send( array( $solo_slot => $merge_raw ) );
					$retry_json = empty( $retry_norm ) ? '' : gida_admin_suite_twilio_content_variables_json( $retry_norm );

					if ( '' !== $retry_json ) {
						$body['ContentVariables'] = $retry_json;
						$r2 = wp_remote_post(
							$twilio_endpoint,
							array(
								'headers' => $post_headers,
								'body'    => $body,
								'timeout' => 12,
							)
						);
						if ( ! is_wp_error( $r2 ) ) {
							$code2 = (int) wp_remote_retrieve_response_code( $r2 );
							$body2 = (string) wp_remote_retrieve_body( $r2 );
							$status = ( 200 === $code2 || 201 === $code2 ) ? 'sent' : 'failed';
							$resp   = $response_notes . '[Attempt 2 merged into slot ' . $solo_slot . '] HTTP ' . $code2 . ' | ' . $body2;
							if ( 'sent' === $status ) {
								$json_vars = $retry_json;
								$norm_vars = $retry_norm;
							}
						} else {
							$wp2 = gida_admin_suite_format_wp_error_for_log( $r2 );
							$resp = $response_notes . '[Attempt 2 transport error] WP_Error: ' . $r2->get_error_message() . ( '' !== $wp2 ? ' | ' . $wp2 : '' );
						}
					}
				}
			}

			if ( 'sent' !== $status ) {
				$rbody   = preg_replace( '/^HTTP\s+\d+\s*\|\s*/', '', (string) $resp );
				$t_check = json_decode( (string) $rbody, true );
				if ( ! is_array( $t_check ) ) {
					$t_check = json_decode( (string) $resp, true );
				}

				$still_bad = strpos( (string) $resp, '21656' ) !== false || ( is_array( $t_check ) && isset( $t_check['code'] ) && (int) $t_check['code'] === 21656 );

				if ( $still_bad && '' !== $json_vars ) {
					delete_transient( $content_var_cache_key );
					if ( '' === trim( (string) $response_notes ) ) {
						$response_notes = '[After first failed POST]' . "\n";
					}
					$response_notes .= "[Post-21656: cleared Twilio Content variable cache & re-resolve keys]\n";

					$res_refetch = gida_admin_suite_fetch_twilio_content_variable_keys( $content_sid_trim );
					$proj_r      = array();
					if ( 'ok' === $res_refetch['status'] && ! empty( $res_refetch['keys'] ) ) {
						$proj_r = gida_admin_suite_map_twilio_content_variables_to_keys(
							$res_refetch['keys'],
							$name_placeholder,
							$services_placeholder
						);
					} else {
						$proj_r = gida_admin_suite_twilio_content_variables_fallback_projection(
							$trigger_key,
							$name_placeholder,
							$services_placeholder
						);
					}

					$norm_r = gida_admin_suite_normalize_twilio_content_variables_for_send( $proj_r );
					$json_r = empty( $norm_r ) ? '' : gida_admin_suite_twilio_content_variables_json( $norm_r );

					if ( '' !== $json_r ) {
						$body['ContentVariables'] = $json_r;
						$r3 = wp_remote_post(
							$twilio_endpoint,
							array(
								'headers' => $post_headers,
								'body'    => $body,
								'timeout' => 12,
							)
						);

						if ( ! is_wp_error( $r3 ) ) {
							$c3        = (int) wp_remote_retrieve_response_code( $r3 );
							$b3        = (string) wp_remote_retrieve_body( $r3 );
							$sent_here = ( 200 === $c3 || 201 === $c3 );
							$status    = $sent_here ? 'sent' : 'failed';
							$resolved  = $res_refetch;
							$norm_vars = $norm_r;
							$json_vars = $json_r;
							$resp      = trim( $response_notes ) . ' [Re-resolve POST] HTTP ' . $c3 . ' | ' . $b3;
						} else {
							$wp3  = gida_admin_suite_format_wp_error_for_log( $r3 );
							$resp = trim( $response_notes ) . ' [Re-resolve POST] WP_Error: ' . $r3->get_error_message() . ( '' !== $wp3 ? ' | ' . $wp3 : '' );
						}
					}
				}
			}
		}
	}

	if ( 'failed' === $status && '' === trim( (string) $resp ) ) {
		if ( ! $sid || ! $token ) {
			$resp = 'Failed: Twilio Account SID or Auth Token missing.';
		} elseif ( '' === $to ) {
			$resp = 'Failed: invalid or empty WhatsApp number after normalization.';
		} elseif ( '' === trim( (string) $template_sid ) ) {
			$resp = 'Failed: empty ContentSid.';
		} else {
			$resp = 'Failed: Twilio request was not sent (precondition).';
		}
	}

	$wpdb->insert(
		$logs_table,
		array(
			'lead_id'            => absint( $lead_id ),
			'phone'              => (string) $to,
			'direction'          => 'outgoing',
			'status'             => $status,
			'trigger_key'        => sanitize_key( (string) $trigger_key ),
			'template_name'      => '',
			'template_sid'       => sanitize_text_field( (string) $template_sid ),
			'content_variables'  => (string) ( '' !== $json_vars ? $json_vars : '' ),
			'message_text'       => '',
			'provider_response'  => (string) $resp,
			'created_at'         => $now,
		),
		array( '%d', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s', '%s' )
	);

	return 'sent' === $status ? 1 : 0;
}

function gida_admin_suite_prepare_twilio_media_url( $media_url ) {
	$url = esc_url_raw( trim( (string) $media_url ) );
	if ( '' === $url ) {
		return '';
	}
	if ( ! wp_http_validate_url( $url ) ) {
		return '';
	}

	$parts = wp_parse_url( $url );
	if ( ! is_array( $parts ) ) {
		return '';
	}

	$scheme = isset( $parts['scheme'] ) ? strtolower( (string) $parts['scheme'] ) : '';
	if ( 'https' !== $scheme ) {
		return '';
	}

	$host = isset( $parts['host'] ) ? strtolower( (string) $parts['host'] ) : '';
	if ( '' === $host ) {
		return '';
	}

	$blocked_hosts = array( 'localhost', '127.0.0.1', '::1' );
	if ( in_array( $host, $blocked_hosts, true ) ) {
		return '';
	}

	return $url;
}

function gida_admin_suite_normalize_whatsapp_to( $phone ) {
	$raw = trim( (string) $phone );
	if ( '' === $raw ) {
		return '';
	}

	$raw = preg_replace( '/\s+/', '', $raw );
	$raw = str_replace( 'whatsapp:', '', $raw );
	$raw = preg_replace( '/[^0-9+]/', '', $raw );
	$raw = ltrim( $raw, '+' );

	// If local Indian 10-digit number is stored, force +91 as requested.
	if ( preg_match( '/^\d{10}$/', $raw ) ) {
		$raw = '91' . $raw;
	}

	if ( '' === $raw ) {
		return '';
	}

	return 'whatsapp:+' . $raw;
}

function gida_admin_suite_sync_booking_subscription( $post_id, $post, $update ) {
	if ( 'gida_booking' !== $post->post_type ) {
		return;
	}
	$email   = (string) get_post_meta( $post_id, '_gida_email', true );
	$name    = (string) get_post_meta( $post_id, '_gida_name', true );
	$service = (string) get_post_meta( $post_id, '_gida_service', true );
	$phone   = (string) get_post_meta( $post_id, '_gida_whatsapp', true );
	if ( '' === $phone ) {
		$phone = (string) get_post_meta( $post_id, '_gida_phone', true );
	}
	if ( ! $email ) {
		return;
	}

	$user = get_user_by( 'email', $email );
	if ( ! $user ) {
		return;
	}

	$history = get_user_meta( $user->ID, '_gida_subscription_history', true );
	if ( ! is_array( $history ) ) {
		$history = array();
	}
	$history[] = array(
		'booking_id' => (int) $post_id,
		'name'       => $name,
		'service'    => $service,
		'phone'      => $phone,
		'date'       => current_time( 'mysql' ),
	);

	update_user_meta( $user->ID, '_gida_subscription_history', array_slice( $history, -50 ) );
	update_user_meta( $user->ID, '_gida_phone', $phone );

	$page_url = gida_admin_suite_map_service_name_to_url( $service );
	if ( $page_url ) {
		update_user_meta( $user->ID, '_gida_subscription_page_url', $page_url );
	}
}
add_action( 'save_post_gida_booking', 'gida_admin_suite_sync_booking_subscription', 20, 3 );

function gida_admin_suite_handle_user_registered_whatsapp_sequence( $user_id ) {
	$user_id = absint( $user_id );
	if ( $user_id <= 0 ) {
		return;
	}
	$user = get_userdata( $user_id );
	if ( ! $user ) {
		return;
	}
	$name  = (string) $user->display_name;
	$phone = (string) get_user_meta( $user_id, '_gida_phone', true );
	if ( '' === trim( $phone ) ) {
		return;
	}
	$sent = gida_admin_suite_trigger_whatsapp( $user_id, $name, $phone, 'user_registered' );
	if ( $sent <= 0 ) {
		gida_admin_suite_trigger_whatsapp_content_template(
			$user_id,
			$phone,
			'user_registered',
			array(
				'name'     => $name,
				'services' => '',
				'amount'   => '',
			)
		);
	}
}
add_action( 'user_register', 'gida_admin_suite_handle_user_registered_whatsapp_sequence', 30 );
